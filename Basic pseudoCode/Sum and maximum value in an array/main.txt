Function Main
    declare integer array score[5]
    declare integer i, ans, max
    set i = 0
    set ans = 0

    for i = 0 to 4
        display"ป้อนตัวเลขตัวที่ " & i + 1
        input score[i]
        set ans = ans + score[i]
    end

    set max = score[0]
    for i = 0 to 4
        if score[i] > max
            set max = score[i]
        end

    end
    display"ผลรวม = " & ans
    display"ค่ามากที่สุด = " & max
End
